```go
// we define the simple fucntions that performs the 
// calculations between the 2 numbers
func add(i int, j int) int { return i + j }
func sub(i int, j int) int { return i - j }
func mul(i int, j int) int { return i * j }
func div(i int, j int) int { return i / j }

// we create map that associates with the operand key with the method
var opMap = map[string]func(int, int) int{
	"+": add,
	"-": sub,
	"*": mul,
	"/": div,
}

func main() {
	// we create some experessions that we can calculate
	expressions := [][]string{
		[]string{"2", "+", "3"},
		[]string{"2", "-", "3"},
		[]string{"2", "*", "3"},
		[]string{"2", "/", "3"},
		[]string{"2", "%", "3"},
		[]string{"two", "+", "three"},
		[]string{"5"},
	}
	// we loop through the expressions objects
	for _, expression := range expressions {
		// if there are no 3 operands in the expression, the expression
		// is invalid by default
		if len(expression) != 3 {
			fmt.Println("invalid expression:", expression)
			continue
		}
		
		// take a note that with atoi we can convert string to integer
		// its very similar to c-style function atoi
		p1, err := strconv.Atoi(expression[0])
		if err != nil {
			// on error with atoi we print the error message and 
			// continue to the next loop
			fmt.Println(err)
			continue
		}
		
		// we take the operand from the expression
		op := expression[1]
		// we associate the expression with the equivalent function from opMap
		opFunc, ok := opMap[op]
		if !ok {
			// in case we dont find the expression on the operand list, 
			// we continue to the next operation element
			fmt.Println("unsupported operator:", op)
			continue
		}
		
		p2, err := strconv.Atoi(expression[2])
		if err != nil {
			fmt.Println(err)
			continue
		}
		// we assign the result of the operated function into result
		result := opFunc(p1, p2)
		// and the print the result
		fmt.Println(result)
	}
}
```
