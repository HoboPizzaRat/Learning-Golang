Cleanup code of things such as network related can be done with defer keyword
- defer code releases resources

example
```go
func main() {
	// we check the program given arguments with lenght of os.Args
	if len(os.Args) < 2 {
		log.Fatal("no file specified")
	}
	// we can acquire program given arguments with os.Args
	f, err := os.Open(os.Args[1])
	if err != nil {
		log.Fatal(err)
	}
	defer f.Close()
	data := make([]byte, 2048)
	for {
		count, err := f.Read(data)
		os.Stdout.Write(data[:count])
		if err != nil {
			if err != io.EOF {
				log.Fatal(err)
			}
			break
		}
	}
}

```
